### 3.2.5.1 Keil-MDK uvision + STC89C52RC REG51定义
```cpp
#include <reg52.h>

// 简单延时函数
void Delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++) {
        for (j = 0; j < 120; j++); // 大约 1ms @ 12MHz
    }
}

void main(void) {
    // 配置 P1.0 为推挽输出（51 默认就是准双向口，可直接输出）
    // 无需额外配置

    while (1) {
        P1_0 = 0;   // 低电平点亮 LED
        Delay_ms(1000);

        P1_0 = 1;   // 高电平熄灭 LED
        Delay_ms(1000);
    }
}
```
